Folding Pathway

The protein starts unfolded (λ < 0, one minimum). As conditions change (temperature drops, pH shifts), λ increases through 0 — the bifurcation point. The landscape splits. The protein must choose: fold correctly or misfold. Misfolding diseases (Alzheimer's, Parkinson's, prions) are proteins that took the wrong branch.

Self objects clone from prototypes. The pitchfork energy is a prototype; folded/misfolded states clone from it.

  parent* = traits clonable.

  " E(x, λ) = x⁴ - λx² "
  energy: x Lambda: lam = (
    (x ^ 4) - (lam * (x ^ 2))
  ).

  " Key identity: E + λ²/4 = (x² - λ/2)² "
  identity: x Lambda: lam = (
    ((self energy: x Lambda: lam) + ((lam ^ 2) / 4))
      = ((x ^ 2) - (lam / 2)) ^ 2
  ).

  " Global minimum energy for λ > 0 "
  minEnergy: lam = ( -1 * ((lam ^ 2) / 4) ).

  " Folded state: x = +√(λ/2) "
  foldedX: lam = ( (lam / 2) sqrt ).

  " Misfolded state: x = -√(λ/2) "
  misfoldedX: lam = ( -1 * ((lam / 2) sqrt) ).

  " Bifurcation: count minima "
  minimaCount: lam = (
    lam < 0 ifTrue: [1] False: [2]
  ).
|)

           folded-x, misfolded-x, minima-count, bifurcation?;
end module;

define class <energy-landscape> (<object>)
  slot lambda :: <real>, required-init-keyword: lambda:;
end class;

// E(x, λ) = x⁴ - λx²
define method pitchfork-energy (x :: <real>, lam :: <real>)
    => (e :: <real>)
  (x ^ 4) - (lam * (x ^ 2))
end method;

// Folded: x = +√(λ/2)
define method folded-x (lam :: <real>) => (x :: <real>)
  sqrt(lam / 2)
end method;

// Misfolded: x = -√(λ/2)
define method misfolded-x (lam :: <real>) => (x :: <real>)
  -sqrt(lam / 2)
end method;

// Minima count: 1 if λ < 0, else 2
define method minima-count (lam :: <real>) => (n :: <integer>)
  if (lam < 0) 1 else 2 end
end method;

// Bifurcation at λ = 0?
define method bifurcation? (lam :: <real>) => (b :: <boolean>)
  lam = 0
end method;

Swift — Type-Safe Bifurcation

Swift structs with value semantics. The energy landscape is a value type; bifurcation is a computed property.

// Swift: pitchfork bifurcation (AppleUXBridge style)

import Foundation

/// Protein folding energy landscape: E(x, λ) = x⁴ - λx²
struct EnergyLandscape {
    let lambda: Double

    /// Energy at position x
    func energy(at x: Double) -> Double {
        pow(x, 4) - lambda * pow(x, 2)
    }

    /// Key identity: E + λ²/4 = (x² - λ/2)²
    func identityHolds(at x: Double) -> Bool {
        let lhs = energy(at: x) + (lambda * lambda) / 4
        let rhs = pow(pow(x, 2) - lambda / 2, 2)
        return abs(lhs - rhs) < 1e-10
    }

    /// Global minimum energy (λ > 0)
    var minEnergy: Double {
        -lambda * lambda / 4
    }

    /// Folded state: x = +√(λ/2)
    var foldedX: Double {
        sqrt(lambda / 2)
    }

    /// Misfolded state: x = -√(λ/2)
    var misfoldedX: Double {
        -sqrt(lambda / 2)
    }

    /// Number of global minima
    var minimaCount: Int {
        lambda < 0 ? 1 : 2
    }

    /// State classification
    var state: FoldingState {
        if lambda < 0 { return .unfolded }
        if lambda == 0 { return .bifurcation }
        return .folded
    }
}

enum FoldingState {
    case unfolded, bifurcation, folded
}

-- Lean 4: ProteinFoldingBifurcation.lean
-- E(x, λ) = x⁴ - λx², proved with pure algebra.

def pitchforkEnergy (x lam : ℝ) : ℝ := x ^ 4 - lam * x ^ 2

-- Key identity: E + λ²/4 = (x² - λ/2)²
theorem pitchfork_identity (x lam : ℝ) :
    pitchforkEnergy x lam + lam ^ 2 / 4 = (x ^ 2 - lam / 2) ^ 2 := by
  unfold pitchforkEnergy
  ring

-- Folded: +√(λ/2) is global min
theorem pitchfork_min_pos {lam : ℝ} (hlam : 0 < lam) :
    pitchforkEnergy (Real.sqrt (lam / 2)) lam = -lam ^ 2 / 4 := by
  -- proof via calc + Real.sq_sqrt

-- Misfolded: -√(λ/2) is global min
theorem pitchfork_min_neg {lam : ℝ} (hlam : 0 < lam) :
    pitchforkEnergy (-Real.sqrt (lam / 2)) lam = -lam ^ 2 / 4 := by
  -- proof via calc + neg_sq + Real.sq_sqrt

-- Bifurcation: minima count changes at λ = 0
theorem pitchfork_bifurcation_count {lam : ℝ} (hlam : 0 < lam) :
    pitchforkEnergy (Real.sqrt (lam / 2)) lam =
      pitchforkEnergy (-Real.sqrt (lam / 2)) lam ∧
    Real.sqrt (lam / 2) ≠ -Real.sqrt (lam / 2) := by
  constructor
  · rw [pitchfork_min_pos hlam, pitchfork_min_neg hlam]
  · exact pitchfork_minima_distinct hlam
